    """MLP escrito do zero em NumPy (Exercícios 2, 3 e 4; também confere o Exercício 1).

Convenções (as mesmas do enunciado):
  * camada l tem pesos W[l] de shape (n_saída, n_entrada) e viés b[l] de shape (n_saída,);
  * com um lote X de shape (N, n_entrada), a pré-ativação é U = X @ W.T + b, ou seja,
    cada linha é u = W x + b para uma amostra;
  * camadas ocultas usam tanh; a ativação da camada de saída vem junto com a perda
    (sigmoide + BCE, softmax + CE, tanh + MSE), porque o gradiente dL/du na saída
    depende do PAR ativação+perda.

A classe MLP NÃO sabe qual problema está resolvendo: tamanho das camadas e perda são
argumentos. forward, backward e step são o mesmo código para binário e multiclasse.
"""
import numpy as np

EPS = 1e-12  # evita log(0) nas entropias cruzadas


# ---------------------------------------------------------------------------
# Ativações
# ---------------------------------------------------------------------------
def tanh(u):
    return np.tanh(u)


def tanh_grad_from_output(h):
    """d tanh(u)/du escrita em função da saída h = tanh(u): 1 - h^2."""
    return 1.0 - h ** 2


def sigmoid(u):
    # forma estável: evita exp de números grandes positivos
    out = np.empty_like(u, dtype=float)
    pos = u >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-u[pos]))
    e = np.exp(u[~pos])
    out[~pos] = e / (1.0 + e)
    return out


def softmax(U):
    # subtrair o máximo da linha não muda o resultado e evita overflow
    E = np.exp(U - U.max(axis=1, keepdims=True))
    return E / E.sum(axis=1, keepdims=True)


# ---------------------------------------------------------------------------
# Perdas: cada uma traz a ativação de saída e dL/du (u = pré-ativação da saída)
# Todas são médias sobre as N amostras do lote, por isso o 1/N em grad_u.
# ---------------------------------------------------------------------------
class BinaryCrossEntropy:
    """Saída sigmoide (1 unidade) + entropia cruzada binária."""

    name = "BCE"

    def encode(self, y):
        return np.asarray(y, dtype=float).reshape(-1, 1)

    def output(self, U):
        return sigmoid(U)

    def value(self, P, Y):
        P = np.clip(P, EPS, 1 - EPS)
        return float(-np.mean(Y * np.log(P) + (1 - Y) * np.log(1 - P)))

    def grad_u(self, P, Y):
        # dL/dp = (p - y) / (p (1 - p)) e dp/du = p (1 - p)  ->  dL/du = p - y
        return (P - Y) / Y.shape[0]

    def predict(self, P):
        return (P[:, 0] >= 0.5).astype(int)


class CategoricalCrossEntropy:
    """Saída softmax (K unidades) + entropia cruzada categórica."""

    name = "CE"

    def __init__(self, n_classes):
        self.n_classes = n_classes

    def encode(self, y):
        return np.eye(self.n_classes)[np.asarray(y, dtype=int)]  # one-hot

    def output(self, U):
        return softmax(U)

    def value(self, P, Y):
        return float(-np.mean(np.sum(Y * np.log(np.clip(P, EPS, 1.0)), axis=1)))

    def grad_u(self, P, Y):
        # derivação no relatório (Exercício 3B): dL/du_k = p_k - y_k
        return (P - Y) / Y.shape[0]

    def predict(self, P):
        return P.argmax(axis=1)


class MeanSquaredErrorTanh:
    """Saída tanh + erro quadrático médio. Usada só para conferir o Exercício 1."""

    name = "MSE"

    def encode(self, y):
        return np.asarray(y, dtype=float).reshape(-1, 1)

    def output(self, U):
        return np.tanh(U)

    def value(self, Yhat, Y):
        return float(np.mean(np.sum((Y - Yhat) ** 2, axis=1)))

    def grad_u(self, Yhat, Y):
        # dL/dŷ = -2 (y - ŷ) / N   e   dŷ/du = 1 - ŷ^2
        return -2.0 * (Y - Yhat) * (1.0 - Yhat ** 2) / Y.shape[0]

    def predict(self, Yhat):
        return Yhat[:, 0]


# ---------------------------------------------------------------------------
# A rede
# ---------------------------------------------------------------------------
class MLP:
    def __init__(self, layer_sizes, loss, rng):
        """layer_sizes = [n_entrada, n_oculta_1, ..., n_saída]; loss = objeto de perda acima."""
        self.sizes = list(layer_sizes)
        self.loss = loss
        self.W, self.b = [], []
        for fan_in, fan_out in zip(self.sizes[:-1], self.sizes[1:]):
            # Xavier/Glorot normal: mantém a variância das ativações tanh entre camadas
            std = np.sqrt(2.0 / (fan_in + fan_out))
            self.W.append(rng.normal(0.0, std, size=(fan_out, fan_in)))
            self.b.append(np.zeros(fan_out))

    @classmethod
    def from_params(cls, W, b, loss):
        """Rede com pesos dados, sem sorteio (usada para conferir o Exercício 1)."""
        net = cls.__new__(cls)
        net.W = [np.array(w, dtype=float) for w in W]
        net.b = [np.array(v, dtype=float) for v in b]
        net.sizes = [net.W[0].shape[1]] + [w.shape[0] for w in net.W]
        net.loss = loss
        return net

    @property
    def n_layers(self):
        return len(self.W)

    def n_params(self):
        return int(sum(W.size + b.size for W, b in zip(self.W, self.b)))

    # ---- forward ----------------------------------------------------------
    def forward(self, X):
        """Propaga X (N, n_entrada) e guarda as ativações de cada camada para o backward."""
        H = X
        self.cache = [H]  # cache[l] = entrada da camada l (cache[0] = X)
        for l in range(self.n_layers):
            U = H @ self.W[l].T + self.b[l]
            last = l == self.n_layers - 1
            H = self.loss.output(U) if last else tanh(U)
            self.cache.append(H)
        return H

    # ---- backward ---------------------------------------------------------
    def backward(self, Y):
        """Gradientes de L em relação a todos os W e b, usando o cache do último forward."""
        gW = [None] * self.n_layers
        gb = [None] * self.n_layers
        delta = self.loss.grad_u(self.cache[-1], Y)  # dL/du da camada de saída, (N, n_saída)
        for l in reversed(range(self.n_layers)):
            H_in = self.cache[l]                     # entrada da camada l, (N, n_entrada)
            gW[l] = delta.T @ H_in                   # dL/dW = soma_n delta_n h_n^T
            gb[l] = delta.sum(axis=0)                # dL/db = soma_n delta_n
            if l > 0:
                dH = delta @ self.W[l]               # dL/dh da camada anterior
                delta = dH * tanh_grad_from_output(H_in)  # dL/du = dL/dh * (1 - h^2)
        return gW, gb

    # ---- update -----------------------------------------------------------
    def step(self, gW, gb, lr):
        """Descida do gradiente: theta <- theta - lr * dL/dtheta."""
        for l in range(self.n_layers):
            self.W[l] -= lr * gW[l]
            self.b[l] -= lr * gb[l]

    # ---- laço de treino ---------------------------------------------------
    def fit(self, X, y, epochs, lr, batch_size, rng):
        """SGD em mini-lotes. Devolve a perda no treino inteiro ao fim de cada época."""
        Y = self.loss.encode(y)
        n = X.shape[0]
        history = []
        for _ in range(epochs):
            order = rng.permutation(n)  # embaralha a cada época
            for start in range(0, n, batch_size):
                idx = order[start:start + batch_size]
                self.forward(X[idx])
                gW, gb = self.backward(Y[idx])
                self.step(gW, gb, lr)
            history.append(self.loss.value(self.forward(X), Y))
        return np.array(history)

    def predict_proba(self, X):
        return self.forward(X)

    def predict(self, X):
        return self.loss.predict(self.forward(X))


# ---------------------------------------------------------------------------
# Utilidades
# ---------------------------------------------------------------------------
def standardize(X_train, *others):
    """z-score com média e desvio do TREINO (sem vazamento do teste)."""
    mu = X_train.mean(axis=0)
    sd = X_train.std(axis=0)
    return [(X_train - mu) / sd] + [(X - mu) / sd for X in others], (mu, sd)


def accuracy(y_true, y_pred):
    return float(np.mean(np.asarray(y_true) == np.asarray(y_pred)))

    Exercício 1¶

    Um passo de backpropagation, à mão. Rede 2–2–1, \(\tanh\) nas duas camadas, perda \(L = (y - \hat y)^2\) com \(N = 1\). A conta abaixo foi feita à mão, com 6 casas decimais em cada valor intermediário (carregando a precisão completa entre os passos). Uso a convenção do enunciado: a linha \(i\) de \(\mathbf{W}^{(1)}\) são os pesos do neurônio oculto \(i\), então \(z^{(1)}_i = \sum_j W^{(1)}_{ij} x_j + b^{(1)}_i\).

    Dados: \(\mathbf{x} = [0.5,\ -0.2]\), \(y = 1\), \(\mathbf{W}^{(1)} = \begin{bmatrix} 0.3 & -0.1 \\ 0.2 & 0.4 \end{bmatrix}\), \(\mathbf{b}^{(1)} = [0.1,\ -0.2]\), \(\mathbf{W}^{(2)} = [0.5,\ -0.3]\), \(b^{(2)} = 0.2\), \(\eta = 0.3\).

    A — Forward pass¶

    1. Pré-ativações ocultas \(\mathbf{z}^{(1)} = \mathbf{W}^{(1)}\mathbf{x} + \mathbf{b}^{(1)}\):

    \[ \begin{aligned} z^{(1)}_1 &= 0.3(0.5) + (-0.1)(-0.2) + 0.1 = 0.15 + 0.02 + 0.1 = \mathbf{0.270000} \\ z^{(1)}_2 &= 0.2(0.5) + 0.4(-0.2) - 0.2 = 0.10 - 0.08 - 0.2 = \mathbf{-0.180000} \end{aligned} \]

    2. Ativações ocultas \(\mathbf{h}^{(1)} = \tanh(\mathbf{z}^{(1)})\):

    \[ h^{(1)}_1 = \tanh(0.27) = \mathbf{0.263625}, \qquad h^{(1)}_2 = \tanh(-0.18) = \mathbf{-0.178081} \]

    3. Pré-ativação da saída \(u^{(2)} = \mathbf{W}^{(2)}\mathbf{h}^{(1)} + b^{(2)}\):

    \[ u^{(2)} = 0.5(0.2636248) + (-0.3)(-0.1780809) + 0.2 = 0.1318124 + 0.0534243 + 0.2 = \mathbf{0.385237} \]

    4. Saída e perda:

    \[ \hat y = \tanh(0.385237) = \mathbf{0.367247}, \qquad L = (1 - 0.367247)^2 = 0.632753^2 = \mathbf{0.400377} \]

    B — Backward pass¶

    Uso \(\frac{d}{du}\tanh(u) = 1 - \tanh^2(u)\), que se escreve com a própria saída: se \(a = \tanh(u)\), então \(\frac{da}{du} = 1 - a^2\).

    1. Da perda até a pré-ativação da saída. \(\dfrac{\partial L}{\partial \hat y} = -2(y - \hat y)\) e \(\dfrac{\partial \hat y}{\partial u^{(2)}} = 1 - \hat y^2\), então

    \[ \begin{aligned} \frac{\partial L}{\partial \hat y} &= -2(1 - 0.367247) = \mathbf{-1.265507} \\ 1 - \hat y^2 &= 1 - 0.134870 = 0.865130 \\ \delta^{(2)} \equiv \frac{\partial L}{\partial u^{(2)}} &= \frac{\partial L}{\partial \hat y}\cdot\frac{\partial \hat y}{\partial u^{(2)}} = (-1.265507)(0.865130) = \mathbf{-1.094828} \end{aligned} \]

    2. Camada de saída. Como \(u^{(2)} = \sum_j W^{(2)}_j h^{(1)}_j + b^{(2)}\), temos \(\partial u^{(2)}/\partial W^{(2)}_j = h^{(1)}_j\) e \(\partial u^{(2)}/\partial b^{(2)} = 1\):

    \[ \begin{aligned} \frac{\partial L}{\partial \mathbf{W}^{(2)}} &= \delta^{(2)}\,\mathbf{h}^{(1)} = [(-1.094828)(0.263625),\ (-1.094828)(-0.178081)] \\ &= \mathbf{[-0.288624,\ 0.194968]} \\ \frac{\partial L}{\partial b^{(2)}} &= \delta^{(2)} = \mathbf{-1.094828} \end{aligned} \]

    3. Propagando para trás. \(\partial u^{(2)}/\partial h^{(1)}_j = W^{(2)}_j\) e \(\partial h^{(1)}_j/\partial z^{(1)}_j = 1 - (h^{(1)}_j)^2\):

    \[ \begin{aligned} \frac{\partial L}{\partial \mathbf{h}^{(1)}} &= \delta^{(2)}\,\mathbf{W}^{(2)} = [(-1.094828)(0.5),\ (-1.094828)(-0.3)] \\ &= \mathbf{[-0.547414,\ 0.328448]} \\ 1 - (\mathbf{h}^{(1)})^2 &= [1 - 0.069498,\ 1 - 0.031713] = [0.930502,\ 0.968287] \\ \delta^{(1)} \equiv \frac{\partial L}{\partial \mathbf{z}^{(1)}} &= \frac{\partial L}{\partial \mathbf{h}^{(1)}} \odot \big(1 - (\mathbf{h}^{(1)})^2\big) = [(-0.547414)(0.930502),\ (0.328448)(0.968287)] \\ &= \mathbf{[-0.509370,\ 0.318032]} \end{aligned} \]

    4. Camada oculta. \(\partial z^{(1)}_i/\partial W^{(1)}_{ij} = x_j\) e \(\partial z^{(1)}_i/\partial b^{(1)}_i = 1\), então \(\partial L/\partial \mathbf{W}^{(1)} = \delta^{(1)}\mathbf{x}^\top\) (produto externo):

    \[ \begin{aligned} \frac{\partial L}{\partial \mathbf{W}^{(1)}} &= \begin{bmatrix} (-0.509370)(0.5) & (-0.509370)(-0.2) \\ (0.318032)(0.5) & (0.318032)(-0.2) \end{bmatrix} = \mathbf{\begin{bmatrix} -0.254685 & 0.101874 \\ 0.159016 & -0.063606 \end{bmatrix}} \\ \frac{\partial L}{\partial \mathbf{b}^{(1)}} &= \delta^{(1)} = \mathbf{[-0.509370,\ 0.318032]} \end{aligned} \]

    \(\theta \leftarrow \theta - \eta\,\partial L/\partial\theta\) com \(\eta = 0.3\). O enunciado fala em "oito parâmetros"; contando escalar por escalar a rede tem nove (4 em \(\mathbf{W}^{(1)}\), 2 em \(\mathbf{b}^{(1)}\), 2 em \(\mathbf{W}^{(2)}\) e 1 em \(b^{(2)}\)). Atualizo todos:

    \[ \begin{aligned} \lVert\nabla L\rVert^2 &= (0.083304 + 0.038012) + 1.198648 + (0.064864 + 0.010378 + 0.025286 + 0.004046) \\ &\quad + (0.259458 + 0.101145) = 1.785141, \end{aligned} \]

    então a previsão linear é \(\Delta L \approx -0.3 \times 1.785141 = -0.535542\). Esse valor é maior que a própria perda (0.400377), ou seja, \(\eta = 0.3\) é um passo grande e a aproximação linear exagera. O sinal, porém, vale para todo passo pequeno o suficiente. Refazendo o forward com os parâmetros novos, à mão:

    \[ \begin{aligned} z^{(1)}_1 &= 0.376405(0.5) + (-0.130562)(-0.2) + 0.252811 = 0.467126 \\ z^{(1)}_2 &= 0.152295(0.5) + 0.419082(-0.2) - 0.295410 = -0.303079 \\ \mathbf{h}^{(1)} &= [\tanh(0.467126),\ \tanh(-0.303079)] = [0.435874,\ -0.294127] \\ u^{(2)} &= 0.586587(0.435874) + (-0.358490)(-0.294127) + 0.528448 = 0.889568 \\ \hat y &= \tanh(0.889568) = 0.711181, \qquad L = (1 - 0.711181)^2 = \mathbf{0.083417} \end{aligned} \]

    A perda cai de 0.400377 para 0.083417. A intuição: como \(\hat y = 0.367 < y = 1\), todo parâmetro se moveu para aumentar \(u^{(2)}\). O viés \(b^{(2)}\) subiu 0.328; \(W^{(2)}_1\) cresceu (porque \(h^{(1)}_1 > 0\)) e \(W^{(2)}_2\) ficou mais negativo (porque \(h^{(1)}_2 < 0\)); na camada oculta, \(h^{(1)}_1\) aumentou e \(h^{(1)}_2\) diminuiu, que é o que os sinais de \(\mathbf{W}^{(2)}\) pedem. No total, \(u^{(2)}\) foi de 0.385 para 0.890.

def finite_difference_max_error(net, h=1e-6):
    """Maior |grad analítico - grad numérico| sobre todos os parâmetros."""
    loss = net.loss
    net.forward(X)
    gW, gb = net.backward(Y)
    worst = 0.0
    for params, grads in ((net.W, gW), (net.b, gb)):
        for P, G in zip(params, grads):
            for idx in np.ndindex(P.shape):
                old = P[idx]
                P[idx] = old + h
                lp = loss.value(net.forward(X), Y)
                P[idx] = old - h
                lm = loss.value(net.forward(X), Y)
                P[idx] = old
                worst = max(worst, abs((lp - lm) / (2 * h) - G[idx]))
    return worst

def best_line_accuracy(X, y, n_angles=720):
    """Melhor acurácia possível com UMA reta: varre direções w e limiares b (w·x >= b -> classe 1)."""
    best = 0.0
    for theta in np.linspace(0, 2 * np.pi, n_angles, endpoint=False):
        s = X @ np.array([np.cos(theta), np.sin(theta)])
        order = np.argsort(s)
        ys = y[order]
        # limiar entre s[k-1] e s[k]: classe 1 para os n-k maiores
        ones_right = np.concatenate([[ys.sum()], ys.sum() - np.cumsum(ys)])
        zeros_left = np.concatenate([[0], np.cumsum(1 - ys)])
        best = max(best, float(((ones_right + zeros_left) / len(y)).max()))
    return best

    C — Implemente o MLP¶

    Implementação (mlp.py). A classe MLP(layer_sizes, loss, rng) recebe a lista de tamanhos das camadas, então serve para qualquer profundidade. Camadas ocultas usam \(\tanh\). A ativação de saída vem junto com a perda, porque o gradiente na saída depende do par. Com um lote \(X\) de \(N\) amostras e \(L\) camadas:

    • forward: \(H^{(0)} = X\), \(\;U^{(l)} = H^{(l-1)}\mathbf{W}^{(l)\top} + \mathbf{b}^{(l)}\), \(\;H^{(l)} = \tanh(U^{(l)})\) para \(l < L\), e \(\hat Y = g(U^{(L)})\), com \(g\) a ativação de saída;
    • perda (binária): saída sigmoide \(p = \sigma(u)\) e entropia cruzada binária \(\mathcal{L} = -\frac{1}{N}\sum_n \big[y_n\log p_n + (1-y_n)\log(1-p_n)\big]\). Como \(\frac{\partial \mathcal{L}}{\partial p} = \frac{1}{N}\frac{p-y}{p(1-p)}\) e \(\sigma'(u) = p(1-p)\), o produto simplifica para \(\frac{\partial \mathcal{L}}{\partial u} = \frac{p - y}{N}\);
    • backward: \(\Delta^{(L)} = \partial\mathcal{L}/\partial U^{(L)}\) (vem da perda) e, para cada camada de trás para frente, \(\;\partial\mathcal{L}/\partial\mathbf{W}^{(l)} = \Delta^{(l)\top}H^{(l-1)}\), \(\;\partial\mathcal{L}/\partial\mathbf{b}^{(l)} = \sum_n \Delta^{(l)}_n\) e \(\;\Delta^{(l-1)} = \big(\Delta^{(l)}\mathbf{W}^{(l)}\big)\odot\big(1 - (H^{(l-1)})^2\big)\). São as mesmas regras do Exercício 1, em forma matricial;
    • atualização: \(\theta \leftarrow \theta - \eta\,\partial\mathcal{L}/\partial\theta\), em mini-lotes embaralhados a cada época.

    Pesos iniciais Xavier/Glorot, \(\mathcal{N}\big(0,\ 2/(n_\text{in}+n_\text{out})\big)\), e vieses zerados. As entradas são padronizadas com média e desvio do treino (sem vazamento do teste).

    D — Análise¶

    A região que o MLP atribui à classe 1 (Figura 3) tem dois pedaços desconexos. São os dois semiplanos além de duas fronteiras quase paralelas, que cortam a diagonal \((t, t)\) em \(t = -1.62\) e \(t = +1.71\). Entre elas fica uma faixa da classe 0 que corre ao longo da antidiagonal \((t, -t)\) e não se fecha: ao longo dela a fronteira nunca é cruzada. Os erros (círculos) estão todos perto das duas fronteiras, onde as gaussianas se sobrepõem.

    Nenhuma reta produz essa região. Uma reta divide o plano em dois semiplanos, e semiplanos são convexos: se o lado "classe 1" contém um ponto perto de \((3,3)\) e outro perto de \((-3,-3)\), ele contém o segmento inteiro entre os dois, incluindo \((0,0)\), o centro da classe 0. O melhor que uma reta consegue é isolar um dos clusters e sacrificar o outro. Varrendo 720 direções e todos os limiares no treino, a melhor reta acerta 74.25%. O MLP resolve o problema porque cada neurônio oculto \(\tanh\) é uma "reta suave". A camada de saída combina essas retas como um OU: classe 1 se o ponto está além da fronteira de cima ou além da de baixo.

    Por que a regressão logística fica abaixo de 50%, e não nos 74% da melhor reta. As duas classes têm praticamente a mesma média. No treino, a classe 0 tem média \((0.02,\ 0.05)\) e a classe 1 tem \((0.21,\ 0.20)\), porque seus dois clusters em \((3,3)\) e \((-3,-3)\) se cancelam. O logit da regressão logística é linear em \(x\), então ela só consegue explorar diferenças de média. Partindo de \(w = 0\), um passo de Newton dá \(w \approx \Sigma^{-1}(\bar m_1 - \bar m_0) = (0.0296,\ 0.0015)\), que é exatamente o que o LogisticRegression devolveu. A diferença entre as médias é pequena e aponta para a diagonal. Ao longo da diagonal, porém, a variância é enorme (autovalores da covariância do treino: 10.93 na direção da diagonal e 1.45 na da antidiagonal), porque é ali que estão os clusters em \(\pm 3\). Por isso \(\Sigma^{-1}\) encolhe essa componente e sobra um vetor minúsculo, quase horizontal e decidido pelo ruído. Sacrificar um cluster inteiro, como faz a melhor reta, custaria uma log-loss enorme (pontos a ~4 unidades do lado errado), e a log-loss prefere ficar "em cima do muro".

    O resultado é a reta tracejada da Figura 3: quase vertical, passando pelo meio da classe 0. No teste, ela põe 47 dos 100 pontos da classe 0 do lado "classe 1", acerta todo o cluster de cima (42 pontos) e erra todo o de baixo (58 pontos). Isso dá \((53 + 42)/200 = 47.5\%\). Ficou abaixo de 50% por amostragem: no treino o cluster de cima tinha mais pontos (208 contra 192), o que inclinou \(w\) levemente a favor dele, e no teste a proporção se inverteu (42 contra 58). Estruturalmente o baseline é uma moeda; o lado em que ela cai é ruído.

    perm = [int(np.argmin([np.abs(np.sort(X[:, j]) - np.sort(Xn[:, k])).max() for k in range(4)]))
            for j in range(4)]
    Xn = Xn[:, perm]
    row_of = {tuple(np.round(r, 9)): i for i, r in enumerate(Xn)}
    idx = np.array([row_of[tuple(np.round(r, 9))] for r in X])
    assert len(set(idx.tolist())) == len(X), "casamento de linhas falhou"
    return idx // (DATA_KW["n_samples"] // 6)  # 250 amostras por cluster


def gaussian_loglik(X, mu, S):
    """log N(x; mu, S) a menos de uma constante (igual para todos os clusters em R^4)."""
    D = X - mu
    _, logdet = np.linalg.slogdet(S)
    return -0.5 * np.einsum("ij,jk,ik->i", D, np.linalg.inv(S), D) - 0.5 * logdet


def cluster_structure(X, y, X_te, y_te, pred_te):
    """Estrutura dos 6 clusters e de onde vêm os erros do MLP.

    Usa o índice de cluster, que o MLP NÃO vê. As gaussianas de cada cluster (média e
    covariância) são estimadas só no treino. Com elas:
      * sobreposição de um par = fração dos pontos dos dois clusters que a própria
        gaussiana do outro cluster explica melhor;
      * "Bayes plug-in" = classificador ideal que conhece as 6 gaussianas
        (prevê o cluster mais verossímil e devolve a classe dele). É o teto de referência.
    """
    cl = recover_clusters(X)
    # mesma chamada da divisão original -> mesmos índices; o assert confere
    X_tr_check, X_te_check, cl_tr, cl_te = train_test_split(X, cl, test_size=0.2, random_state=42, stratify=y)
    assert np.array_equal(X_te_check, X_te)
    cls_of = np.arange(6) % N_CLASSES
    mus = [X_tr_check[cl_tr == k].mean(axis=0) for k in range(6)]
    covs = [np.cov(X_tr_check[cl_tr == k].T) for k in range(6)]

    pairs = []
    for a in range(6):
        for b in range(a + 1, 6):
            if cls_of[a] == cls_of[b]:
                continue
            m = (cl == a) | (cl == b)
            closer_to_a = gaussian_loglik(X[m], mus[a], covs[a]) > gaussian_loglik(X[m], mus[b], covs[b])
            pairs.append({
                "clusters": [a, b], "classes": [int(cls_of[a]), int(cls_of[b])],
                "distance": float(np.linalg.norm(mus[b] - mus[a])),
                # centros ~ vértices (±1, ±1, ±1, ±1) do hipercubo: em quantas coordenadas diferem
                "n_coords_differ": int(np.sum(np.sign(mus[a]) != np.sign(mus[b]))),
                "overlap": float(np.mean(np.where(closer_to_a, a, b) != cl[m])),
            })
    pairs.sort(key=lambda p: -p["overlap"])

    per_cluster = []
    for k in range(6):
        m = cl_te == k
        per_cluster.append({
            "cluster": k, "class": int(cls_of[k]), "n_test": int(m.sum()),
            "predicted_counts": np.bincount(pred_te[m], minlength=N_CLASSES).tolist(),
            "errors": int(np.sum(pred_te[m] != y_te[m])),
            # desvios-padrão ao longo dos eixos principais (a transformação aleatória deforma cada cluster)
            "principal_stds": np.round(np.sqrt(np.linalg.eigvalsh(covs[k])), 2).tolist(),
        })

    LL = np.column_stack([gaussian_loglik(X_te, mus[k], covs[k]) for k in range(6)])
    bayes_pred = cls_of[LL.argmax(axis=1)]
    bayes_cm = confusion_matrix(y_te, bayes_pred)
    flipped = int(sum(np.sum(y[cl == k] != cls_of[k]) for k in range(6)))
    return {
        "centroids": np.round(mus, 3).tolist(),
        "cross_class_pairs": pairs,
        "test_errors_by_cluster": per_cluster,
        "bayes_plugin_test_accuracy": accuracy(y_te, bayes_pred),
        "bayes_plugin_confusion_matrix": bayes_cm.tolist(),
        "labels_flipped_by_flip_y": flipped,
    }

    B — Adapte a rede¶

    As mudanças ficam só na saída e na perda: 3 unidades de saída, softmax e entropia cruzada categórica (CategoricalCrossEntropy(3) em mlp.py).

    Derivação de \(\partial L/\partial u\) para softmax + entropia cruzada. Para uma amostra com rótulo one-hot \(\mathbf{y}\) (\(\sum_k y_k = 1\)) e pré-ativações de saída \(\mathbf{u}\):

    \[ p_k = \frac{e^{u_k}}{\sum_j e^{u_j}}, \qquad L = -\sum_k y_k \log p_k . \]

    Jacobiano da softmax. Pela regra do quociente, se \(k = j\): \(\frac{\partial p_k}{\partial u_k} = \frac{e^{u_k}\sum_j e^{u_j} - e^{u_k}e^{u_k}}{(\sum_j e^{u_j})^2} = p_k(1 - p_k)\). Se \(k \ne j\): \(\frac{\partial p_k}{\partial u_j} = \frac{-e^{u_k}e^{u_j}}{(\sum_i e^{u_i})^2} = -p_k p_j\). Nos dois casos,

    \[ \frac{\partial p_k}{\partial u_j} = p_k(\delta_{kj} - p_j). \]

    Regra da cadeia. \(\frac{\partial L}{\partial p_k} = -\frac{y_k}{p_k}\), então

    \[ \frac{\partial L}{\partial u_j} = \sum_k \frac{\partial L}{\partial p_k}\frac{\partial p_k}{\partial u_j} = -\sum_k \frac{y_k}{p_k}\,p_k(\delta_{kj} - p_j) = -\sum_k y_k\delta_{kj} + p_j\sum_k y_k = -y_j + p_j \cdot 1 = \boxed{\,p_j - y_j\,} \]

    Para a média num lote de \(N\) amostras, \(\partial \mathcal{L}/\partial U = (P - Y)/N\).

    Por que fica tão simples. O \(1/p_k\) que vem da derivada do log cancela exatamente o fator \(p_k\) do Jacobiano da softmax, e a restrição \(\sum_k y_k = 1\) elimina o resto. É o mesmo cancelamento da sigmoide com a BCE no Exercício 2, onde \(\partial\mathcal{L}/\partial u = p - y\). Nos dois casos o gradiente na saída é o erro de previsão: fica em \([-1, 1]\), é grande quando o modelo está confiante e errado, e não some quando a saída satura, ao contrário do que aconteceria com MSE sobre a softmax. A sigmoide + BCE é o caso \(K = 2\) da softmax + CE.

def train(layers, seed, data, epochs=EPOCHS, eval_every=None):
    """Treina uma rede com np.random.default_rng(seed) (inicialização + embaralhamento).

    Com eval_every, treina em blocos e mede a acurácia de teste a cada bloco. Isso não
    muda o treino: fit consome o rng só com uma permutação por época, então 10 blocos de
    30 épocas = 1 bloco de 300 épocas, bit a bit. As métricas "no orçamento" são tiradas
    exatamente na época EPOCHS.
    """
    X_tr, X_te, y_tr, y_te = data
    rng = np.random.default_rng(seed)
    net = MLP(layers, CategoricalCrossEntropy(ex3.N_CLASSES), rng)
    step = eval_every or epochs
    history, checkpoints, at_budget = [], [], None
    done = 0
    while done < epochs:
        history.extend(net.fit(X_tr, y_tr, epochs=step, lr=LR, batch_size=BATCH, rng=rng))
        done += step
        checkpoints.append((done, accuracy(y_te, net.predict(X_te))))
        if done == EPOCHS:
            pred = net.predict(X_te)
            at_budget = {
                "final_train_loss": float(history[-1]),
                "train_accuracy": accuracy(y_tr, net.predict(X_tr)),
                "test_accuracy": accuracy(y_te, pred),
                "confusion_matrix": confusion_matrix(y_te, pred),
            }
    return {"n_params": net.n_params(), "history": np.array(history),
            "checkpoints": checkpoints, "at_budget": at_budget}

    shallow = [train(SHALLOW, s, data, LONG_EPOCHS, EVAL_EVERY) for s in SEEDS]
    deep = [train(DEEP, s, data, LONG_EPOCHS, EVAL_EVERY) for s in SEEDS]
    fig6_loss(shallow, deep)
    fig_c1_long(shallow, deep)
    s_sum, d_sum = summarize(shallow), summarize(deep)
    diffs = np.array(d_sum["test_accuracy"]["per_seed"]) - np.array(s_sum["test_accuracy"]["per_seed"])
    p = s_sum["test_accuracy"]["mean"]

    # robustez do "par mais confundido" do Exercício 3 entre seeds (rede rasa)
    def pair_counts(cm):
        return {f"{i}-{j}": int(cm[i, j] + cm[j, i]) for i in range(3) for j in range(i + 1, 3)}
    shallow_pairs = [pair_counts(r["at_budget"]["confusion_matrix"]) for r in shallow]

    # C — controle: mesma contagem de parâmetros, profundidade diferente
    matched = {}
    for label, (wide, narrow_deep) in MATCHED.items():
        matched[label] = {}
        for layers in (wide, narrow_deep):
            runs = [train(layers, s, data) for s in SEEDS]
            matched[label]["-".join(map(str, layers))] = {"n_params": runs[0]["n_params"], **summarize(runs)}

    # diagnóstico do treino estendido: acurácia média de teste em algumas épocas
    def acc_at(runs, epoch):
        return float(np.mean([dict(r["checkpoints"])[epoch] for r in runs]))
    long_view = {str(e): {"shallow_test_acc": acc_at(shallow, e), "deep_test_acc": acc_at(deep, e),
                          "shallow_train_loss": float(np.mean([r["history"][e - 1] for r in shallow])),
                          "deep_train_loss": float(np.mean([r["history"][e - 1] for r in deep]))}
                 for e in (50, 100, 200, 300, 500, 1000)}

    print(f"[Ex4] principal {DEEP}: teste {main_run['test_accuracy']:.4f} | "
          f"seeds {SEEDS}: raso {s_sum['test_accuracy']['mean']:.4f} x fundo {d_sum['test_accuracy']['mean']:.4f} "
          f"(diferença média {diffs.mean():+.4f})")
    return {
        "main_run": main_run,
        "seeds": list(SEEDS),
        "shallow": {"layers": SHALLOW, "n_params": shallow[0]["n_params"], **s_sum},
        "deep": {"layers": DEEP, "n_params": deep[0]["n_params"], **d_sum},
        "test_acc_diff_deep_minus_shallow": {
            "per_seed": diffs.tolist(), "mean": float(diffs.mean()), "std": float(diffs.std(ddof=1)),
            "seeds_where_deep_wins": int(np.sum(diffs > 0)),
        },
        "test_accuracy_binomial_se": float(np.sqrt(p * (1 - p) / len(y_te))),
        "shallow_pair_confusions_per_seed": shallow_pairs,
        "parameter_matched": matched,
        "long_training": long_view,
    }

    C — Análise¶

    A perda de treino e a acurácia de teste contam histórias diferentes. Pela perda de treino (Figura 6), a rede funda é muito melhor: 20% menor em 300 épocas (0.233 contra 0.291) e metade em 1000 épocas (0.124 contra 0.248). A acurácia de teste mal acompanha: +1.5 ponto em 300 épocas, e em 1000 épocas a rede funda já está abaixo da rasa (83.40% contra 83.87%) enquanto continua baixando a perda de treino. A distância entre treino e teste cresce (acurácia de treino 89.50% → 91.27%; diferença treino–teste de 4.90 para 5.20 pontos). Essa queda extra da perda de treino é a rede memorizando os pontos da região de sobreposição. Eles são irredutíveis: até o classificador ideal do Exercício 3 erra 10.7% do teste.

    O tamanho do ganho também pede cautela. 1.47 ponto são ~4 dos 300 pontos de teste, e o erro-padrão binomial de uma acurácia nesse teste é \(\sqrt{p(1-p)/300} \approx\) 2.08 pontos. As seeds controlam a variância da inicialização, mas não a do próprio conjunto de teste.

    O que a segunda camada acrescenta, em termos de representação, e o que custa. Os dados são 6 nuvens gaussianas em vértices de um hipercubo 4D. Separar nuvens assim pede poucas fronteiras aproximadamente planas, uma ou duas por par de clusters vizinhos, e uma camada oculta com 16 unidades \(\tanh\) já tem mais "retas suaves" do que o problema pede. O teorema de aproximação universal garante que uma camada basta em princípio, e aqui ela basta na prática. O que limita a acurácia é a sobreposição entre C1 e seus vizinhos (teto de ~89%), não a expressividade da rede. Uma segunda camada compõe funções das features da primeira ("regiões feitas de regiões") e pode representar certas funções com exponencialmente menos unidades. Esses dados, porém, não têm estrutura hierárquica para ser composta. Quanto ao custo de otimização, ele não apareceu aqui: com só duas camadas \(\tanh\), inicialização Xavier e SGD em mini-lotes, os gradientes não somem, e a rede funda até converge mais rápido por época (Figura 6). O custo apareceu como variância: mais parâmetros (403 para 1200 exemplos), mais sobreajuste quando o treino continua, e desvio entre seeds maior (0.93 contra 0.77 ponto em 300 épocas).

    Que dados fariam a profundidade valer a pena. Dados com estrutura composicional, em que a fronteira é feita de pedaços que se repetem ou se aninham. O exemplo formal é a função "serra" obtida compondo \(k\) vezes uma função tenda: ela tem \(2^k\) dentes e sai de uma rede com \(O(k)\) camadas estreitas, mas uma rede rasa precisa de largura exponencial em \(k\) para aproximá-la (Telgarsky, 2016). Na prática: espirais entrelaçadas com muitas voltas, tabuleiros de xadrez com muitas casas e, sobretudo, imagens, som e texto, em que bordas formam texturas, texturas formam partes e partes formam objetos. Também ajudaria ter muito mais exemplos por parâmetro, para que a capacidade extra fosse usada para modelar a fronteira e não o ruído. Com 6 nuvens gaussianas em 4D e 1200 pontos, a resposta honesta é que a profundidade em si não ajudou: o ganho medido some quando se iguala o número de parâmetros e se inverte quando se treina por mais tempo.
